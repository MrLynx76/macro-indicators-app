    <script>

        let password = null;

        function login() {

            const input = document.getElementById('passwordInput');

            password = input.value;


            if (!password) {

                alert('Ingresa contraseña');

                return;

            }


            document.getElementById('loginScreen').style.display = 'none';

            document.getElementById('dashboard').style.display = 'block';

            fetchData();

        }


        function logout() {

            password = null;

            document.getElementById('loginScreen').style.display = 'flex';

            document.getElementById('dashboard').style.display = 'none';

            document.getElementById('passwordInput').value = '';

        }


        async function fetchData() {

            try {

                const response = await fetch('/api/data', {

                    headers: {

                        'X-Password': password

                    }

                });


                if (response.status === 401) {

                    logout();

                    return;

                }


                currentData = await response.json();

                renderIndicators();

                document.getElementById('errorMessage').innerHTML = '';

            } catch (error) {

                document.getElementById('errorMessage').innerHTML = `<div class="error">Error: ${error.message}</div>`;

            }

        }


        function renderIndicators() {

            const container = document.getElementById('indicatorsContainer');

            container.innerHTML = '';


            for (const [code, data] of Object.entries(currentData)) {

                const statusClass = data.status || 'gray';

                const html = `

                    <div class="indicator-card">

                        <div class="indicator-header">

                            <div class="indicator-name">${data.name}</div>

                            <div class="status-badge ${statusClass}"></div>

                        </div>

                        <div class="current-value">${data.current !== null ? data.current.toFixed(2) : 'N/A'}</div>

                        <div class="last-date">Último: ${data.date}</div>


                        <div class="last-three">

                            <strong>Últimos 3 valores:</strong>

                            ${data.last_three.map(v => `

                                <div class="last-three-item">

                                    <span>${v.date}</span>

                                    <span>${v.value.toFixed(2)}</span>

                                </div>

                            `).join('')}

                        </div>


                        <div class="chart-container">

                            <canvas id="chart-${code}"></canvas>

                        </div>

                    </div>

                `;

                container.innerHTML += html;

            }


            for (const [code, data] of Object.entries(currentData)) {

                if (data.year_data && data.year_data.length > 0) {

                    renderChart(code, data);

                }

            }

        }


        function renderChart(code, data) {

            const ctx = document.getElementById(`chart-${code}`).getContext('2d');


            if (charts[code]) {

                charts[code].destroy();

            }


            charts[code] = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: data.year_data.map(d => d.date),

                    datasets: [{

                        label: data.name,

                        data: data.year_data.map(d => d.value),

                        borderColor: '#007bff',

                        backgroundColor: 'rgba(0, 123, 255, 0.1)',

                        tension: 0.1,

                        fill: true

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    plugins: {

                        legend: {

                            display: false

                        }

                    },

                    scales: {

                        y: {

                            beginAtZero: false,

                            grid: {

                                color: '#333'

                            },

                            ticks: {

                                color: '#888'

                            }

                        },

                        x: {

                            grid: {

                                display: false

                            },

                            ticks: {

                                color: '#888',

                                maxRotation: 45

                            }

                        }

                    }

                }

            });

        }

        function closeThresholds() {

            document.getElementById('thresholdsModal').style.display = 'none';

        }


        async function saveThresholds() {

            const inputs = document.querySelectorAll('#thresholdsForm input');

            const thresholds = {};


            inputs.forEach(input => {

                const indicator = input.dataset.indicator;

                const type = input.dataset.type;

                const value = parseFloat(input.value);


                if (!thresholds[indicator]) {

                    thresholds[indicator] = { normal: [0, 0], tension: [0, 0], crisis: [0, 0] };

                }


                const [status, minmax] = type.split('_');

                const index = minmax === 'min' ? 0 : 1;

                thresholds[indicator][status][index] = value;

            });


            try {

                const response = await fetch('/api/thresholds', {

                    method: 'POST',

                    headers: {

                        'Content-Type': 'application/json',

                        'X-Password': password

                    },

                    body: JSON.stringify(thresholds)

                });


                if (response.ok) {

                    closeThresholds();

                    fetchData();

                    alert('Umbrales guardados');

                }

            } catch (error) {

                alert('Error guardando umbrales: ' + error.message);

            }

        }


        document.getElementById('passwordInput').addEventListener('keypress', (e) => {

            if (e.key === 'Enter') login();

        });
